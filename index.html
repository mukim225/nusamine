Firestore Security Rules (Anti-Peretasan)

Tempel aturan ini ke Firebase Console → Firestore → Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Aturan Keamanan NusaMine Production
    match /artifacts/{appId}/users/{userId}/{document=**} {
      // Pengguna hanya boleh membaca & mengubah datanya sendiri
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    match /artifacts/{appId}/public/data/appSettings/{docId} {
      // Semua user terautentikasi bisa membaca kurs & pengaturan
      allow read: if request.auth != null;
      // Hanya email Owner yang boleh mengubah pengaturan global
      allow write: if request.auth != null
                   && request.auth.token.email == "owner@nusamine.app";
    }

    match /artifacts/{appId}/public/data/withdrawals/{wdId} {
      // User terautentikasi bisa mengajukan penarikan & melihat riwayat
      allow read, create: if request.auth != null;
      // Hanya Owner yang boleh menyetujui atau menolak penarikan
      allow update, delete: if request.auth != null
                            && request.auth.token.email == "owner@nusamine.app";
    }
  }
}
